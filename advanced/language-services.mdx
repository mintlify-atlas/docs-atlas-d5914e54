# Language Services

VS Code's language services provide IntelliSense, validation, navigation, and refactoring features. Understanding the language service architecture is essential for implementing language support or extending existing languages.

## Architecture Overview

<Info>
**Language Service Components:**
- **Language Service** (`ILanguageService`): Core registry for language definitions
- **Language Features**: Provider-based APIs for editor features
- **Tokenization**: Syntax highlighting via TextMate grammars or Monarch
- **Language Configurations**: Bracket matching, auto-closing, indentation rules
</Info>

**Key Properties:**
- **comments**: Line and block comment tokens
- **brackets**: Bracket pairs for matching
- **autoClosingPairs**: Auto-close pairs when typing
- **surroundingPairs**: Pairs for surround-with commands
- **folding**: Code folding patterns
- **wordPattern**: Regex defining word boundaries
- **indentationRules**: Auto-indentation rules
- **onEnterRules**: Behavior when pressing Enter

## Diagnostics (Validation)

```typescript
import * as vscode from 'vscode';

const diagnosticCollection = vscode.languages.createDiagnosticCollection('mylang');

function validateDocument(document: vscode.TextDocument): void {
  const diagnostics: vscode.Diagnostic[] = [];
  const text = document.getText();
  
  // Example: Check for TODO comments
  const todoPattern = /TODO:/gi;
  let match;
  
  while ((match = todoPattern.exec(text)) !== null) {
    const startPos = document.positionAt(match.index);
    const endPos = document.positionAt(match.index + match[0].length);
    
    const diagnostic = new vscode.Diagnostic(
      new vscode.Range(startPos, endPos),
      'TODO comment found',
      vscode.DiagnosticSeverity.Information
    );
    diagnostic.code = 'todo-comment';
    diagnostic.source = 'mylang';
    
    diagnostics.push(diagnostic);
  }
  
  // Example: Check for unused variables
  const unusedPattern = /const\s+(\w+)\s*=/gi;
  while ((match = unusedPattern.exec(text)) !== null) {
    const varName = match[1];
    if (!new RegExp(`\\b${varName}\\b`, 'g').test(text.slice(match.index + match[0].length))) {
      const startPos = document.positionAt(match.index + match[0].indexOf(varName));
      const endPos = document.positionAt(match.index + match[0].indexOf(varName) + varName.length);
      
      const diagnostic = new vscode.Diagnostic(
        new vscode.Range(startPos, endPos),
        `Variable '${varName}' is never used`,
        vscode.DiagnosticSeverity.Warning
      );
      diagnostic.tags = [vscode.DiagnosticTag.Unnecessary];
      
      diagnostics.push(diagnostic);
    }
  }
  
  diagnosticCollection.set(document.uri, diagnostics);
}

// Validate on document change
vscode.workspace.onDidChangeTextDocument(e => {
  if (e.document.languageId === 'mylang') {
    validateDocument(e.document);
  }
});

// Validate on document open
vscode.workspace.onDidOpenTextDocument(document => {
  if (document.languageId === 'mylang') {
    validateDocument(document);
  }
});
```

<Warning>
**Performance Considerations:**
- Debounce validation for large documents
- Use `CancellationToken` to cancel long operations
- Implement incremental validation when possible
- Clear diagnostics when documents close
</Warning>

class MyCodeActionProvider implements vscode.CodeActionProvider {

      command: 'mylang.extractFunction',

  }

## Language Server Protocol (LSP)

For complex language support, implement a Language Server:

### Language Server Structure

```typescript
import {
  createConnection,
  TextDocuments,
  ProposedFeatures,
  InitializeParams,
  TextDocumentSyncKind
} from 'vscode-languageserver/node';
import { TextDocument } from 'vscode-languageserver-textdocument';

// Create connection
const connection = createConnection(ProposedFeatures.all);
const documents = new TextDocuments(TextDocument);

// Initialize
connection.onInitialize((params: InitializeParams) => {
  return {
    capabilities: {
      textDocumentSync: TextDocumentSyncKind.Incremental,
      completionProvider: {
        resolveProvider: true,
        triggerCharacters: ['.', ':']
      },
      hoverProvider: true,
      definitionProvider: true,
      referencesProvider: true,
      documentSymbolProvider: true,
      codeActionProvider: true,
      documentFormattingProvider: true
    }
  };
});

// Completion
connection.onCompletion((params) => {
  const document = documents.get(params.textDocument.uri);
  // Return completion items
  return [];
});

// Hover
connection.onHover((params) => {
  const document = documents.get(params.textDocument.uri);
  // Return hover info
  return null;
});

// Start listening
documents.listen(connection);
connection.listen();
```

### Language Client

```typescript
import * as path from 'path';
import { workspace, ExtensionContext } from 'vscode';
import {
  LanguageClient,
  LanguageClientOptions,
  ServerOptions,
  TransportKind
} from 'vscode-languageclient/node';

let client: LanguageClient;

export function activate(context: ExtensionContext) {
  const serverModule = context.asAbsolutePath(
    path.join('server', 'out', 'server.js')
  );
  
  const serverOptions: ServerOptions = {
    run: { module: serverModule, transport: TransportKind.ipc },
    debug: {
      module: serverModule,
      transport: TransportKind.ipc,
      options: { execArgv: ['--nolazy', '--inspect=6009'] }
    }
  };
  
  const clientOptions: LanguageClientOptions = {
    documentSelector: [{ scheme: 'file', language: 'mylang' }],
    synchronize: {
      fileEvents: workspace.createFileSystemWatcher('**/.mylangrc')
    }
  };
  
  client = new LanguageClient(
    'mylangServer',
    'MyLang Language Server',
    serverOptions,
    clientOptions
  );
  
  client.start();
}

export function deactivate(): Thenable<void> | undefined {
  if (!client) {
    return undefined;
  }
  return client.stop();
}
```
