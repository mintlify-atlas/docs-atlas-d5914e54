## Overview

VS Code's architecture is organized into four distinct layers, each building upon the layers below it. This strict layering ensures clear separation of concerns, maintainability, and the ability to reuse components across different contexts.

```
┌─────────────────────────────────────────┐
│         workbench (Application)         │  ← Full VS Code experience
├─────────────────────────────────────────┤
│         editor (Monaco Editor)          │  ← Standalone text editor
├─────────────────────────────────────────┤
│        platform (Core Services)         │  ← Service infrastructure
├─────────────────────────────────────────┤
│       base (Foundation Utilities)       │  ← Basic primitives
└─────────────────────────────────────────┘
```

<Note>
Layers can only import from layers below them, never above. This is enforced by the `npm run valid-layers-check` command.
</Note>

## Layer 1: Base

Location: `src/vs/base/`

The **base layer** provides fundamental utilities and abstractions with zero dependencies on other VS Code layers. This layer is purely about providing building blocks.

### Key Components

<Tabs>
  <Tab title="Lifecycle">
    Resource management through the disposable pattern:
    
    ```typescript
    // From src/vs/base/common/lifecycle.ts
    export interface IDisposable {
      dispose(): void;
    }
    
    export class DisposableStore implements IDisposable {
      private readonly _disposables = new Set<IDisposable>();
      
      public add<T extends IDisposable>(disposable: T): T {
        this._disposables.add(disposable);
        return disposable;
      }
      
      public dispose(): void {
        for (const disposable of this._disposables) {
          disposable.dispose();
        }
        this._disposables.clear();
      }
    }
    
    export abstract class Disposable implements IDisposable {
      private readonly _store = new DisposableStore();
      
      protected _register<T extends IDisposable>(t: T): T {
        return this._store.add(t);
      }
      
      dispose(): void {
        this._store.dispose();
      }
    }
    ```
    
    **Usage**: All long-lived objects that manage resources should extend `Disposable` or use `DisposableStore`.
  </Tab>
  
  <Tab title="Events">
    Type-safe event system for reactive programming:
    
    ```typescript
    // From src/vs/base/common/event.ts
    export interface Event<T> {
      (listener: (e: T) => unknown, thisArgs?: any, disposables?: IDisposable[]): IDisposable;
    }
    
    export class Emitter<T> {
      private readonly _event: Event<T>;
      
      get event(): Event<T> {
        return this._event;
      }
      
      fire(event: T): void {
        // Notify all listeners
      }
    }
    
    // Usage example
    class MyClass {
      private readonly _onDidChange = new Emitter<string>();
      readonly onDidChange: Event<string> = this._onDidChange.event;
      
      private notifyChange(value: string): void {
        this._onDidChange.fire(value);
      }
    }
    
    // Subscribing to events
    const disposable = myClass.onDidChange(value => {
      console.log('Changed:', value);
    });
    // Later: disposable.dispose();
    ```
    
    **Key Features**:
    - Automatic memory management via disposables
    - Event composition (map, filter, debounce)
    - Once, any, signal patterns
  </Tab>
  
  <Tab title="Async Utilities">
    Utilities for asynchronous operations:
    
    ```typescript
    // From src/vs/base/common/async.ts
    
    // Debouncing
    export class Delayer<T> {
      constructor(defaultDelay: number);
      trigger(task: () => Promise<T>, delay?: number): Promise<T>;
    }
    
    // Throttling
    export class ThrottledDelayer<T> extends Delayer<T> {
      // Ensures task runs at most once per interval
    }
    
    // Run once per scheduler tick
    export class RunOnceScheduler {
      constructor(runner: () => void, delay: number);
      schedule(): void;
      cancel(): void;
    }
    
    // Sequential task queue
    export class Queue<T> {
      queue<T>(task: () => Promise<T>): Promise<T>;
    }
    
    // Usage example
    class AutoSave {
      private readonly saveDelayer = new Delayer<void>(1000);
      
      triggerSave(): void {
        this.saveDelayer.trigger(() => this.doSave());
      }
      
      private async doSave(): Promise<void> {
        // Save logic
      }
    }
    ```
  </Tab>
  
  <Tab title="Data Structures">
    Specialized data structures for common patterns:
    
    ```typescript
    // Linked list for efficient insertion/removal
    export class LinkedList<T> {
      push(element: T): () => void;
      unshift(element: T): () => void;
      // Returns removal function
    }
    
    // Resource map with URI keys

      set(key: URI, value: T): void;

      // Handles URI comparison correctly
    }
    
    // Lazy value initialization

      constructor(private readonly _getValue: () => T);
      get value(): T {
        // Computed only once
      }
    }
    ```
  </Tab>
  
  <Tab title="UI Components">
    Basic UI building blocks (in `src/vs/base/browser/`):
    
    - **Tree**: Virtual scrolling tree component
    - **List**: High-performance list rendering
    - **Scrollbar**: Custom scrollbar implementation
    - **SplitView**: Resizable split panels
    - **Grid**: 2D grid layout
    - **DOM utilities**: Safe DOM manipulation
    
    These are low-level components used by higher layers.
  </Tab>
</Tabs>

## Layer 2: Platform

Location: `src/vs/platform/`

The **platform layer** provides core services and the dependency injection infrastructure. This layer defines service interfaces and their implementations.

### Dependency Injection

The instantiation service is the heart of the platform layer:

```typescript
// From src/vs/platform/instantiation/common/instantiation.ts
export interface IInstantiationService {
  createInstance<T>(ctor: new (...args: any[]) => T): T;
  invokeFunction<R>(fn: (accessor: ServicesAccessor) => R): R;
  createChild(services: ServiceCollection): IInstantiationService;
}

// Service decorator
export function createDecorator<T>(serviceId: string): ServiceIdentifier<T>;

// Example service definition
export const IConfigurationService = createDecorator<IConfigurationService>('configurationService');

export interface IConfigurationService {
  readonly _serviceBrand: undefined;
  getValue<T>(key: string): T;
  updateValue(key: string, value: any): Promise<void>;
}
```

See [Dependency Injection](/architecture/dependency-injection) for complete details.

### Service Registration

```typescript
// From src/vs/platform/instantiation/common/extensions.ts
export const enum InstantiationType {
  Eager = 0,    // Instantiate immediately
  Delayed = 1   // Instantiate on first use (preferred)
}

export function registerSingleton<T>(
  id: ServiceIdentifier<T>,
  ctor: new (...args: any[]) => T,
  instantiationType: InstantiationType
): void;

// Example registration
registerSingleton(
  IConfigurationService,
  ConfigurationService,
  InstantiationType.Delayed
);
```

<Note>
Prefer `InstantiationType.Delayed` to reduce startup time. Only use `Eager` for services that must initialize before any other code runs.
</Note>

Location: `src/vs/editor/`

The **editor layer** implements the Monaco Editor, VS Code's standalone text editor. This layer can be used independently of the workbench.

## Layer 4: Workbench

Location: `src/vs/workbench/`

The **workbench layer** provides the complete VS Code application experience, built on top of the editor.

### Workbench Structure

```
src/vs/workbench/
├── browser/           # Core workbench UI
│   ├── workbench.ts   # Main workbench class
│   ├── layout.ts      # Layout management
│   └── parts/         # UI parts (sidebar, panel, etc.)
├── services/          # Workbench services
│   ├── textfile/      # Text file handling
│   ├── extensions/    # Extension management
│   ├── debug/         # Debug service
│   └── search/        # Search service
├── contrib/           # Feature contributions
│   ├── files/         # File explorer
│   ├── search/        # Search view
│   ├── debug/         # Debug view
│   ├── terminal/      # Integrated terminal
│   └── git/           # Git integration
└── api/               # Extension API implementation
    └── common/        # VS Code API
```

### Workbench Layout

The workbench is divided into parts:

```
┌─────────────────────────────────────────────┐
│              Title Bar                       │
├─────────┬───────────────────────┬────────────┤
│ Activity│                       │            │
│  Bar    │   Editor Area         │  Sidebar   │
│         │                       │            │
├─────────┴───────────────────────┴────────────┤
│              Panel                           │
├──────────────────────────────────────────────┤
│              Status Bar                      │
└──────────────────────────────────────────────┘
```

## Layer Validation

VS Code enforces layer dependencies:

```bash
# Check layer violations
npm run valid-layers-check
```

Common violations:
- Base importing from platform
- Editor importing from workbench
- Common importing from browser/node

<Warning>
Always run layer validation before submitting pull requests. Layer violations will fail CI builds.
</Warning>
