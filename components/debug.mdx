### IDebugService Interface

```typescript
export interface IDebugService {
    readonly _serviceBrand: undefined;
    
    // State
    readonly state: State;
    readonly onDidChangeState: Event<State>;
    
    // Sessions
    readonly onDidNewSession: Event<IDebugSession>;
    readonly onWillNewSession: Event<IDebugSession>;
    readonly onDidEndSession: Event<{ session: IDebugSession; restart: boolean }>;
    
    getModel(): IDebugModel;
    getViewModel(): IViewModel;
    getConfigurationManager(): IConfigurationManager;
    
    // Session control
    startDebugging(
        launch: ILaunch | undefined,
        configOrName?: IConfig | string,
        options?: IDebugSessionOptions
    ): Promise<boolean>;
    
    restartSession(session: IDebugSession): Promise<void>;
    stopSession(session: IDebugSession): Promise<void>;
    
    // Breakpoints
    addBreakpoints(uri: URI, breakpoints: IBreakpointData[]): Promise<IBreakpoint[]>;
    removeBreakpoints(id?: string): Promise<void>;
    enableOrDisableBreakpoints(enable: boolean, breakpoint?: IEnablement): Promise<void>;
}
```

### Using Debug Service

```typescript
import { IDebugService, State } from 'vs/workbench/contrib/debug/common/debug';

export class DebugManager {
    constructor(
        @IDebugService private readonly debugService: IDebugService
    ) {
        // Monitor debug state
        this.debugService.onDidChangeState(state => {
            switch (state) {
                case State.Inactive:
                    console.log('No debugging');
                    break;
                case State.Initializing:
                    console.log('Starting debug session');
                    break;
                case State.Running:
                    console.log('Debugger running');
                    break;
                case State.Stopped:
                    console.log('Debugger stopped at breakpoint');
                    break;
            }
        });
        
        // Monitor sessions
        this.debugService.onDidNewSession(session => {
            console.log('New session:', session.name);
        });
    }
    
    async startDebugging(): Promise<void> {
        const success = await this.debugService.startDebugging(
            undefined,
            'Launch Program'
        );
        
        if (!success) {
            console.error('Failed to start debugging');
        }
    }
}
```

### IDebugSession Interface

```typescript
export interface IDebugSession {
    readonly id: string;
    readonly name: string;
    readonly state: State;
    readonly configuration: IConfig;
    
    // Lifecycle
    initialize(dbgr: IDebugger): Promise<void>;
    launchOrAttach(config: IConfig): Promise<void>;
    restart(): Promise<void>;
    terminate(restart?: boolean): Promise<void>;
    disconnect(restart?: boolean): Promise<void>;
    
    // Execution control
    continue(threadId: number): Promise<void>;
    pause(threadId: number): Promise<void>;
    stepOver(threadId: number): Promise<void>;
    stepInto(threadId: number): Promise<void>;
    stepOut(threadId: number): Promise<void>;
    
    // Stack and variables
    stackTrace(threadId: number): Promise<IStackFrame[]>;
    scopes(frameId: number): Promise<IScope[]>;
    variables(variablesReference: number): Promise<IVariable[]>;
    
    // Evaluation
    evaluate(expression: string, frameId?: number): Promise<IExpression>;
}
```

### Working with Debug Sessions

```typescript
import { IDebugService } from 'vs/workbench/contrib/debug/common/debug';

export class DebugController {
    constructor(
        @IDebugService private readonly debugService: IDebugService
    ) {}
    
    async pauseExecution(): Promise<void> {
        const session = this.debugService.getViewModel().focusedSession;
        if (session) {
            const thread = this.debugService.getViewModel().focusedThread;
            if (thread) {
                await session.pause(thread.threadId);
            }
        }
    }
    
    async stepOver(): Promise<void> {
        const session = this.debugService.getViewModel().focusedSession;
        const thread = this.debugService.getViewModel().focusedThread;
        
        if (session && thread) {
            await session.stepOver(thread.threadId);
        }
    }
    
    async evaluateExpression(expression: string): Promise<string> {
        const session = this.debugService.getViewModel().focusedSession;
        const frame = this.debugService.getViewModel().focusedStackFrame;
        
        if (session && frame) {
            const result = await session.evaluate(expression, frame.frameId);
            return result.value;
        }
        
        return '';
    }
}
```

### Managing Breakpoints

```typescript
import { IDebugService } from 'vs/workbench/contrib/debug/common/debug';
import { URI } from 'vs/base/common/uri';

export class BreakpointManager {
    constructor(
        @IDebugService private readonly debugService: IDebugService
    ) {}
    
    async addBreakpoint(uri: URI, lineNumber: number): Promise<void> {
        await this.debugService.addBreakpoints(uri, [
            {
                lineNumber,
                enabled: true
            }
        ]);
    }
    
    async addConditionalBreakpoint(
        uri: URI,
        lineNumber: number,
        condition: string
    ): Promise<void> {
        await this.debugService.addBreakpoints(uri, [
            {
                lineNumber,
                enabled: true,
                condition
            }
        ]);
    }
    
    async addLogpoint(
        uri: URI,
        lineNumber: number,
        message: string
    ): Promise<void> {
        await this.debugService.addBreakpoints(uri, [
            {
                lineNumber,
                enabled: true,
                logMessage: message
            }
        ]);
    }
    
    async removeAllBreakpoints(): Promise<void> {
        await this.debugService.removeBreakpoints();
    }
    
    async toggleBreakpoint(uri: URI, lineNumber: number): Promise<void> {
        const model = this.debugService.getModel();
        const existing = model.getBreakpoints({ uri, lineNumber });
        
        if (existing.length > 0) {
            await this.debugService.removeBreakpoints(existing[0].getId());
        } else {
            await this.addBreakpoint(uri, lineNumber);
        }
    }
}
```

### Creating a Debug Adapter

```typescript
import { DebugSession } from '@vscode/debugadapter';
import { DebugProtocol } from '@vscode/debugprotocol';

export class MyDebugSession extends DebugSession {
    protected initializeRequest(
        response: DebugProtocol.InitializeResponse,
        args: DebugProtocol.InitializeRequestArguments
    ): void {
        response.body = {
            supportsConfigurationDoneRequest: true,
            supportsEvaluateForHovers: true,
            supportsStepBack: false,
            supportsConditionalBreakpoints: true,
            supportsLogPoints: true
        };
        this.sendResponse(response);
    }
    
    protected launchRequest(
        response: DebugProtocol.LaunchResponse,
        args: DebugProtocol.LaunchRequestArguments
    ): void {
        // Start the debuggee
        this.sendResponse(response);
    }
    
    protected setBreakPointsRequest(
        response: DebugProtocol.SetBreakpointsResponse,
        args: DebugProtocol.SetBreakpointsArguments
    ): void {
        const breakpoints = args.breakpoints || [];
        // Set breakpoints in debuggee
        
        response.body = {
            breakpoints: breakpoints.map(bp => ({
                verified: true,
                line: bp.line
            }))
        };
        this.sendResponse(response);
    }
}
```

## Best Practices

<Warning>
**Session Lifecycle:** Always properly terminate debug sessions. Lingering sessions can hold onto resources.
</Warning>

<Tip>
**Conditional Breakpoints:** Use expressions that can be quickly evaluated. Complex conditions slow down debugging.
</Tip>

<Note>
**Debug Adapters:** Test your debug adapter with various configurations. Handle protocol errors gracefully.
</Note>
