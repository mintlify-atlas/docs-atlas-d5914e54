## Overview

The Editor component (`src/vs/editor/`) is the Monaco Editor - a powerful, feature-rich code editor that powers VS Code. It provides syntax highlighting, IntelliSense, bracket matching, diff viewing, and countless other features.

**Location:** `src/vs/editor/`

### ITextModel Interface

**Location:** `src/vs/editor/common/model.ts:500`

Represents the document content:

```typescript
export interface ITextModel {
    readonly uri: URI;
    readonly id: string;
    
    // Content access
    getValue(): string;
    getLineCount(): number;
    getLineContent(lineNumber: number): string;
    getValueInRange(range: IRange): string;
    
    // Editing
    applyEdits(operations: IIdentifiedSingleEditOperation[]): void;
    pushEditOperations(
        beforeCursorState: Selection[] | null,
        editOperations: IIdentifiedSingleEditOperation[],
        cursorStateComputer: ICursorStateComputer | null
    ): Selection[] | null;
    
    // Events
    onDidChangeContent(listener: (e: IModelContentChangedEvent) => void): IDisposable;
}
```

### Creating a Text Model

```typescript
import { IModelService } from 'vs/editor/common/services/model';
import { URI } from 'vs/base/common/uri';

export class DocumentManager {
    constructor(
        @IModelService private readonly modelService: IModelService
    ) {}
    
    createDocument(uri: URI, content: string, languageId: string): ITextModel {
        return this.modelService.createModel(
            content,
            { languageId },
            uri
        );
    }
    
    getDocument(uri: URI): ITextModel | null {
        return this.modelService.getModel(uri);
    }
}
```

### Editing Text Models

```typescript
import { Range } from 'vs/editor/common/core/range';
import { ITextModel } from 'vs/editor/common/model';

function editDocument(model: ITextModel): void {
    // Simple edit
    model.applyEdits([{
        range: new Range(1, 1, 1, 1),
        text: '// New comment\n'
    }]);
    
    // Multiple edits with cursor tracking
    const edits = [
        {
            range: new Range(1, 1, 1, 5),
            text: 'replaced'
        },
        {
            range: new Range(3, 1, 3, 1),
            text: 'inserted'
        }
    ];
    
    model.pushEditOperations(
        null,
        edits,
        () => null // Cursor state computer
    );
}
```

### Creating an Editor

import { ICodeEditorService } from 'vs/editor/browser/services/codeEditorService';

            this.container,

        return editor;
    }
}

### Hover Provider Example

```typescript
languages.HoverProviderRegistry.register('javascript', {
    provideHover(
        model: ITextModel,
        position: Position,
        token: CancellationToken
    ): languages.ProviderResult<languages.Hover> {
        const word = model.getWordAtPosition(position);
        if (!word) return null;
        
        return {
            contents: [
                { value: `**${word.word}**` },
                { value: 'Documentation for this symbol' }
            ],
            range: {
                startLineNumber: position.lineNumber,
                startColumn: word.startColumn,
                endLineNumber: position.lineNumber,
                endColumn: word.endColumn
            }
        };
    }
});
```

### Creating an Editor Contribution

```typescript
import { ICodeEditor } from 'vs/editor/browser/editorBrowser';
import { IEditorContribution } from 'vs/editor/common/editorCommon';
import { Disposable } from 'vs/base/common/lifecycle';

class MyEditorContribution extends Disposable implements IEditorContribution {
    public static readonly ID = 'editor.contrib.myContribution';
    
    constructor(
        private readonly editor: ICodeEditor
    ) {
        super();
        
        // Listen to editor events
        this._register(editor.onDidChangeModel(() => {
            this.onModelChanged();
        }));
        
        this._register(editor.onDidChangeCursorPosition(e => {
            this.onCursorPositionChanged(e);
        }));
    }
    
    private onModelChanged(): void {
        const model = this.editor.getModel();
        if (model) {
            // React to model changes
        }
    }
    
    private onCursorPositionChanged(e: ICursorPositionChangedEvent): void {
        console.log('Cursor moved to:', e.position);
    }
}

// Register the contribution
import { registerEditorContribution } from 'vs/editor/browser/editorExtensions';

registerEditorContribution(
    MyEditorContribution.ID,
    MyEditorContribution,
    EditorContributionInstantiation.AfterFirstRender
);
```

## Editor Commands and Actions

**Location:** `src/vs/editor/browser/editorExtensions.ts`

### Registering Editor Actions

```typescript
import { EditorAction, registerEditorAction } from 'vs/editor/browser/editorExtensions';
import { ICodeEditor } from 'vs/editor/browser/editorBrowser';
import { EditorContextKeys } from 'vs/editor/common/editorContextKeys';
import { KeyCode, KeyMod } from 'vs/base/common/keyCodes';

class MyEditorAction extends EditorAction {
    constructor() {
        super({
            id: 'myExtension.myAction',
            label: 'My Action',
            alias: 'My Action',
            precondition: EditorContextKeys.editorTextFocus,
            kbOpts: {
                kbExpr: EditorContextKeys.editorTextFocus,
                primary: KeyMod.CtrlCmd | KeyCode.KeyK,
                weight: 100
            }
        });
    }
    
    run(accessor: ServicesAccessor, editor: ICodeEditor): void {
        const model = editor.getModel();
        if (!model) return;
        
        const selection = editor.getSelection();
        if (selection) {
            // Perform action
            console.log('Selected text:', model.getValueInRange(selection));
        }
    }
}

registerEditorAction(MyEditorAction);
```

## Best Practices

<Warning>
**Model Lifecycle:** Always dispose text models when no longer needed. Use `IModelService` to create models for automatic lifecycle management.
</Warning>

<Tip>
**Editor Performance:** For large files, consider using `readOnly` mode or disabling expensive features like minimap temporarily.
</Tip>

<Note>
**Language Features:** Register language features early and make providers efficient. They're called frequently during editing.
</Note>
