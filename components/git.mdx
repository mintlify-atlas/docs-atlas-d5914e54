## Overview

The SCM (Source Control Management) component (`src/vs/workbench/contrib/scm/`) provides VS Code's Git integration and source control UI. It includes the SCM viewlet, repository management, and extensible SCM provider support.

**Location:** `src/vs/workbench/contrib/scm/`

### Using SCM Service

```typescript
import { ISCMService, ISCMRepository } from 'vs/workbench/contrib/scm/common/scm';

export class SCMManager {
    constructor(
        @ISCMService private readonly scmService: ISCMService
    ) {
        // Monitor repositories
        this.scmService.onDidAddRepository(repo => {
            console.log('Repository added:', repo.provider.label);
        });
        
        this.scmService.onDidRemoveRepository(repo => {
            console.log('Repository removed:', repo.provider.label);
        });
    }
    
    getAllRepositories(): ISCMRepository[] {
        return this.scmService.repositories;
    }
    
    getGitRepositories(): ISCMRepository[] {
        return this.scmService.repositories.filter(
            repo => repo.provider.id === 'git'
        );
    }
}
```

## SCM Providers {#providers}

### Creating an SCM Provider

```typescript
import { ISCMService, ISCMProvider, ISCMResourceGroup, ISCMResource } from 'vs/workbench/contrib/scm/common/scm';
import { URI } from 'vs/base/common/uri';
import { Emitter, Event } from 'vs/base/common/event';

class MySCMProvider implements ISCMProvider {
    readonly id = 'myScm';
    readonly label = 'My SCM';
    readonly rootUri: URI;
    
    private _onDidChangeResources = new Emitter<void>();
    readonly onDidChangeResources: Event<void> = this._onDidChangeResources.event;
    
    private _groups: ISCMResourceGroup[] = [];
    
    constructor(rootUri: URI) {
        this.rootUri = rootUri;
        this.createGroups();
    }
    
    get groups(): ISequence<ISCMResourceGroup> {
        return {
            elements: this._groups,
            onDidSplice: Event.None
        };
    }
    
    private createGroups(): void {
        this._groups = [
            {
                id: 'workingTree',
                label: 'Changes',
                hideWhenEmpty: false,
                resources: [],
                onDidChange: Event.None
            },
            {
                id: 'index',
                label: 'Staged Changes',
                hideWhenEmpty: true,
                resources: [],
                onDidChange: Event.None
            }
        ];
    }
    
    async getOriginalResource(uri: URI): Promise<URI | null> {
        // Return URI to original (HEAD) version
        return uri.with({ scheme: 'myScm-original' });
    }
    
    dispose(): void {
        this._onDidChangeResources.dispose();
    }
}

// Register provider
export class MySCMExtension {
    constructor(
        @ISCMService private readonly scmService: ISCMService
    ) {}
    
    activate(rootUri: URI): void {
        const provider = new MySCMProvider(rootUri);
        const repository = this.scmService.registerSCMProvider(provider);
        
        // Provider is now active
    }
}
```

```typescript
export interface ISCMInput {
    // Value
    value: string;
    readonly onDidChange: Event<string>;
    
    // Placeholder
    placeholder: string;
    readonly onDidChangePlaceholder: Event<string>;
    
    // Validation
    validateInput: ISCMInputValidator | undefined;
    
    // Visibility
    visible: boolean;
    readonly onDidChangeVisibility: Event<boolean>;
    
    // Focus
    readonly onDidChangeFocus: Event<void>;
}

export interface ISCMInputValidator {
    (value: string, cursorPosition: number): Promise<ISCMInputValidation | undefined>;
}

export interface ISCMInputValidation {
    message: string;
    type: InputValidationType;
}

export enum InputValidationType {
    Information = 0,
    Warning = 1,
    Error = 2
}
```

### Using SCM Input

```typescript
import { ISCMRepository } from 'vs/workbench/contrib/scm/common/scm';

function setupInput(repository: ISCMRepository): void {
    const input = repository.input;
    
    // Set placeholder
    input.placeholder = 'Message (Ctrl+Enter to commit)';
    
    // Validate input
    input.validateInput = async (value, cursorPosition) => {
        if (value.length === 0) {
            return {
                message: 'Commit message cannot be empty',
                type: InputValidationType.Error
            };
        }
        
        if (value.length > 72) {
            return {
                message: 'First line should be 72 characters or less',
                type: InputValidationType.Warning
            };
        }
        
        return undefined;
    };
    
    // Listen to changes
    input.onDidChange(value => {
        console.log('Commit message:', value);
    });
}
```

### Custom SCM Actions

```typescript
import { registerAction2, Action2 } from 'vs/platform/actions/common/actions';
import { ISCMService } from 'vs/workbench/contrib/scm/common/scm';

class CommitAllAction extends Action2 {
    constructor() {
        super({
            id: 'myExtension.commitAll',
            title: { value: 'Commit All Repos', original: 'Commit All Repos' },
            category: { value: 'SCM', original: 'SCM' },
            f1: true
        });
    }
    
    async run(accessor: ServicesAccessor): Promise<void> {
        const scmService = accessor.get(ISCMService);
        
        for (const repo of scmService.repositories) {
            const message = repo.input.value;
            if (message && repo.provider.acceptInputCommand) {
                // Execute commit command
                await accessor.get(ICommandService).executeCommand(
                    repo.provider.acceptInputCommand.id,
                    ...repo.provider.acceptInputCommand.arguments || []
                );
            }
        }
    }
}

registerAction2(CommitAllAction);
```

## Best Practices

<Warning>
**Resource Updates:** Batch resource group updates for better performance. Frequent updates can cause UI flicker.
</Warning>

<Tip>
**Original Resources:** Implement `getOriginalResource` efficiently. It's called frequently for diff views.
</Tip>

<Note>
**Input Validation:** Keep validation fast. It runs on every keystroke and shouldn't block the UI.
</Note>

## Extension Integration

### SCM Provider Extension

```typescript
// Extension API for SCM providers
import * as vscode from 'vscode';

class MySCMProvider implements vscode.SourceControlProvider {
    readonly id = 'myScm';
    readonly label = 'My SCM';
    
    private _sourceControl: vscode.SourceControl;
    private _changesGroup: vscode.SourceControlResourceGroup;
    private _stagedGroup: vscode.SourceControlResourceGroup;
    
    constructor(private readonly workspaceRoot: vscode.Uri) {
        // Create source control
        this._sourceControl = vscode.scm.createSourceControl(
            this.id,
            this.label,
            workspaceRoot
        );
        
        // Create resource groups
        this._changesGroup = this._sourceControl.createResourceGroup(
            'changes',
            'Changes'
        );
        
        this._stagedGroup = this._sourceControl.createResourceGroup(
            'staged',
            'Staged Changes'
        );
        
        // Set accept input command
        this._sourceControl.acceptInputCommand = {
            command: 'myExtension.commit',
            title: 'Commit'
        };
    }
    
    async refresh(): Promise<void> {
        // Update resource groups
        const files = await this.getChangedFiles();
        
        this._changesGroup.resourceStates = files.map(file => ({
            resourceUri: file.uri,
            decorations: {
                strikeThrough: file.deleted,
                faded: file.ignored,
                letter: file.status,
                color: this.getStatusColor(file.status)
            }
        }));
    }
}
```
