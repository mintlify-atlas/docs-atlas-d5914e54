## Overview

Language Features provide the rich editing experience in VS Code, including IntelliSense, error checking, code formatting, and more. These features are powered by language servers, built-in providers, and extensions.

**Locations:**
- `src/vs/editor/common/languages/` - Language feature registries
- `src/vs/editor/contrib/` - Editor language features
- `extensions/` - Built-in language extensions

### Completion Provider

```typescript
import * as languages from 'vs/editor/common/languages';
import { ITextModel } from 'vs/editor/common/model';
import { Position } from 'vs/editor/common/core/position';
import { CancellationToken } from 'vs/base/common/cancellation';

// Register completion provider
languages.CompletionProviderRegistry.register('typescript', {
    provideCompletionItems(
        model: ITextModel,
        position: Position,
        context: languages.CompletionContext,
        token: CancellationToken
    ): languages.ProviderResult<languages.CompletionList> {
        const word = model.getWordUntilPosition(position);
        const range = {
            startLineNumber: position.lineNumber,
            endLineNumber: position.lineNumber,
            startColumn: word.startColumn,
            endColumn: word.endColumn
        };
        
        return {
            suggestions: [
                {
                    label: 'myFunction',
                    kind: languages.CompletionItemKind.Function,
                    insertText: 'myFunction($0)',
                    insertTextRules: languages.CompletionItemInsertTextRule.InsertAsSnippet,
                    detail: '(property) myFunction(): void',
                    documentation: 'Does something useful',
                    range,
                    sortText: '0'
                },
                {
                    label: 'myVariable',
                    kind: languages.CompletionItemKind.Variable,
                    insertText: 'myVariable',
                    detail: 'const myVariable: string',
                    range
                }
            ]
        };
    },
    
    // Optional: Resolve additional details
    resolveCompletionItem(
        item: languages.CompletionItem,
        token: CancellationToken
    ): languages.ProviderResult<languages.CompletionItem> {
        // Load additional documentation or details
        return item;
    }
});
```

## Diagnostics {#diagnostics}

**Location:** `src/vs/editor/common/model/`

### Markers (Diagnostics)

```typescript
import { IMarkerService, IMarkerData, MarkerSeverity } from 'vs/platform/markers/common/markers';
import { URI } from 'vs/base/common/uri';
import { Range } from 'vs/editor/common/core/range';

export class DiagnosticsProvider {
    constructor(
        @IMarkerService private readonly markerService: IMarkerService
    ) {}
    
    setDiagnostics(uri: URI, diagnostics: IMarkerData[]): void {
        this.markerService.changeOne('myExtension', uri, diagnostics);
    }
    
    addError(uri: URI, message: string, line: number, column: number): void {
        const marker: IMarkerData = {
            severity: MarkerSeverity.Error,
            message,
            startLineNumber: line,
            startColumn: column,
            endLineNumber: line,
            endColumn: column + 1,
            source: 'myLinter'
        };
        
        const existing = this.markerService.read({ resource: uri });
        this.markerService.changeOne('myExtension', uri, [...existing, marker]);
    }
    
    clearDiagnostics(uri: URI): void {
        this.markerService.changeOne('myExtension', uri, []);
    }
    
    getAllDiagnostics(): IMarkerData[] {
        return this.markerService.read();
    }
}
```

## Code Actions {#code-actions}

**Location:** `src/vs/editor/contrib/codeAction/`

### Code Action Provider

```typescript
languages.CodeActionProviderRegistry.register('typescript', {
    provideCodeActions(
        model: ITextModel,
        range: Range | Selection,
        context: languages.CodeActionContext,
        token: CancellationToken
    ): languages.ProviderResult<languages.CodeActionList> {
        const actions: languages.CodeAction[] = [];
        
        // Quick fix for errors
        if (context.only === languages.CodeActionKind.QuickFix) {
            for (const marker of context.markers) {
                if (marker.message.includes('unused')) {
                    actions.push({
                        title: 'Remove unused variable',
                        kind: languages.CodeActionKind.QuickFix,
                        diagnostics: [marker],
                        edit: {
                            edits: [{
                                resource: model.uri,
                                versionId: model.getVersionId(),
                                textEdit: {
                                    range: marker,
                                    text: ''
                                }
                            }]
                        }
                    });
                }
            }
        }
        
        // Refactoring actions
        if (context.only === languages.CodeActionKind.Refactor) {
            actions.push({
                title: 'Extract to function',
                kind: languages.CodeActionKind.Refactor,
                command: {
                    id: 'myExtension.extractFunction',
                    title: 'Extract Function',
                    arguments: [model.uri, range]
                }
            });
        }
        
        return {
            actions,
            dispose: () => {}
        };
    }
});
```

```typescript
languages.DocumentFormattingEditProviderRegistry.register('typescript', {
    displayName: 'TypeScript Formatter',
```

## Navigation Features

### Go to Definition

```typescript
languages.DefinitionProviderRegistry.register('typescript', {
    provideDefinition(
        model: ITextModel,
        position: Position,
        token: CancellationToken
    ): languages.ProviderResult<languages.Definition> {
        const word = model.getWordAtPosition(position);
        if (!word) return null;
        
        // Find definition
        const definition = this.findDefinition(word.word);
        
        if (definition) {
            return {
                uri: definition.uri,
                range: definition.range
            };
        }
        
        return null;
    }
});
```

### Find References

```typescript
languages.ReferenceProviderRegistry.register('typescript', {
    provideReferences(
        model: ITextModel,
        position: Position,
        context: languages.ReferenceContext,
        token: CancellationToken
    ): languages.ProviderResult<languages.Location[]> {
        const word = model.getWordAtPosition(position);
        if (!word) return [];
        
        // Find all references
        const references = this.findReferences(word.word);
        
        return references.map(ref => ({
            uri: ref.uri,
            range: ref.range
        }));
    }
});
```

## Language Server Protocol

**Location:** `extensions/*/client/` and `extensions/*/server/`

### LSP Client

```typescript
import {
    LanguageClient,
    LanguageClientOptions,
    ServerOptions,
    TransportKind
} from 'vscode-languageclient/node';

let client: LanguageClient;

export function activate(context: vscode.ExtensionContext) {
    // Server options
    const serverModule = context.asAbsolutePath(
        path.join('server', 'out', 'server.js')
    );
    
    const serverOptions: ServerOptions = {
        run: {
            module: serverModule,
            transport: TransportKind.ipc
        },
        debug: {
            module: serverModule,
            transport: TransportKind.ipc,
            options: { execArgv: ['--nolazy', '--inspect=6009'] }
        }
    };
    
    // Client options
    const clientOptions: LanguageClientOptions = {
        documentSelector: [
            { scheme: 'file', language: 'typescript' },
            { scheme: 'file', language: 'javascript' }
        ],
        synchronize: {
            fileEvents: vscode.workspace.createFileSystemWatcher('**/*.{ts,js}')
        }
    };
    
    // Create and start client
    client = new LanguageClient(
        'myLanguageServer',
        'My Language Server',
        serverOptions,
        clientOptions
    );
    
    client.start();
}

export function deactivate(): Thenable<void> | undefined {
    if (!client) {
        return undefined;
    }
    return client.stop();
}
```

## Rename Provider

```typescript
languages.RenameProviderRegistry.register('typescript', {
    provideRenameEdits(
        model: ITextModel,
        position: Position,
        newName: string,
        token: CancellationToken
    ): languages.ProviderResult<languages.WorkspaceEdit & languages.Rejection> {
        const word = model.getWordAtPosition(position);
        if (!word) return { rejectReason: 'No symbol at position' };
        
        // Find all occurrences
        const edits = this.findAllOccurrences(word.word);
        
        return {
            edits: edits.map(edit => ({
                resource: edit.uri,
                versionId: undefined,
                textEdit: {
                    range: edit.range,
                    text: newName
                }
            }))
        };
    },
    
    resolveRenameLocation(
        model: ITextModel,
        position: Position,
        token: CancellationToken
    ): languages.ProviderResult<languages.RenameLocation & languages.Rejection> {
        const word = model.getWordAtPosition(position);
        if (!word) return { rejectReason: 'No symbol at position' };
        
        return {
            range: new Range(
                position.lineNumber,
                word.startColumn,
                position.lineNumber,
                word.endColumn
            ),
            text: word.word
        };
    }
});
```

## Best Practices

<Warning>
**Performance:** Language providers are called frequently. Keep them fast and cancel early when token is cancelled.
</Warning>

<Tip>
**Progressive Enhancement:** Return basic results quickly, then use resolve methods for expensive operations.
</Tip>

<Note>
**Language Servers:** Use LSP for complex language features. It provides better isolation and multi-workspace support.
</Note>
