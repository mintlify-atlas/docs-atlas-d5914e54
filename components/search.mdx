### Search Queries

```typescript
import { ISearchService, ITextQuery, QueryType } from 'vs/workbench/services/search/common/search';
import { URI } from 'vs/base/common/uri';

export class SearchManager {
    constructor(
        @ISearchService private readonly searchService: ISearchService
    ) {}
    
    async searchText(pattern: string, folder: URI): Promise<void> {
        const query: ITextQuery = {
            type: QueryType.Text,
            contentPattern: {
                pattern,
                isRegExp: false,
                isCaseSensitive: false,
                isWordMatch: false
            },
            folderQueries: [
                {
                    folder,
                    excludePattern: {
                        '**/node_modules': true,
                        '**/dist': true
                    },
                    includePattern: {
                        '**/*.ts': true,
                        '**/*.js': true
                    }
                }
            ],
            maxResults: 10000
        };
        
        const result = await this.searchService.textSearch(
            query,
            undefined,
            (progress) => {
                console.log('Search progress:', progress);
            }
        );
        
        console.log('Search complete:', result.results.length, 'results');
        console.log('Stats:', result.stats);
    }
    
    async searchFiles(pattern: string, folder: URI): Promise<void> {
        const query: IFileQuery = {
            type: QueryType.File,
            filePattern: pattern,
            folderQueries: [{ folder }],
            maxResults: 500
        };
        
        const result = await this.searchService.fileSearch(query);
        
        for (const match of result.results) {
            console.log('File:', match.resource.fsPath);
        }
    }
}
```

## Query Builder {#query-builder}

**Location:** `src/vs/workbench/services/search/common/queryBuilder.ts`

### QueryBuilder Class

```typescript
import { QueryBuilder } from 'vs/workbench/services/search/common/queryBuilder';
import { IConfigurationService } from 'vs/platform/configuration/common/configuration';
import { IWorkspaceContextService } from 'vs/platform/workspace/common/workspace';

export class SearchQueryManager {
    private queryBuilder: QueryBuilder;
    
    constructor(
        @IConfigurationService configurationService: IConfigurationService,
        @IWorkspaceContextService workspaceContextService: IWorkspaceContextService
    ) {
        this.queryBuilder = this.instantiationService.createInstance(QueryBuilder);
    }
    
    buildTextQuery(options: ITextQueryBuilderOptions): ITextQuery {
        return this.queryBuilder.text(
            {
                pattern: options.pattern,
                isRegExp: options.isRegExp,
                isCaseSensitive: options.isCaseSensitive,
                isWordMatch: options.isWordMatch
            },
            options.folderResources || [],
            {
                excludePattern: options.excludePattern,
                includePattern: options.includePattern,
                maxResults: options.maxResults,
                previewOptions: options.previewOptions
            }
        );
    }
}
```

### Replace Operations

```typescript
import { IReplaceService } from 'vs/workbench/contrib/search/browser/replace';
import { ISearchTreeFileMatch, ISearchTreeMatch } from 'vs/workbench/contrib/search/browser/searchTreeModel/searchTreeCommon';

export class ReplaceManager {
    constructor(
        @IReplaceService private readonly replaceService: IReplaceService
    ) {}
    
    async replaceSingleMatch(match: ISearchTreeMatch): Promise<void> {
        await this.replaceService.replace(match);
    }
    
    async replaceAllInFile(fileMatch: ISearchTreeFileMatch): Promise<void> {
        await this.replaceService.replace([fileMatch]);
    }
    
    async previewReplace(
        match: ISearchTreeMatch,
        sideBySide: boolean = false
    ): Promise<void> {
        await this.replaceService.openReplacePreview(
            match,
            false,
            sideBySide,
            true
        );
    }
    
    setReplaceText(text: string): void {
        this.replaceService.updateReplaceString(text);
    }
}
```

### Search Result Classes

```typescript
export class SearchResult implements ISearchResult {
    private _folderMatches: Map<URI, FolderMatch> = new Map();
    
    get folderMatches(): ISearchTreeFolderMatch[] {
        return Array.from(this._folderMatches.values());
    }
    
    get count(): number {
        return this.folderMatches.reduce((sum, fm) => sum + fm.count, 0);
    }
    
    add(raw: IFileMatch, searchInstanceID: string): void {
        // Add file match to appropriate folder
    }
    
    clear(): void {
        this._folderMatches.clear();
    }
}

export class FileMatch implements ISearchTreeFileMatch {
    private _matches: Map<string, Match> = new Map();
    
    constructor(
        public readonly resource: URI,
        private readonly parent: FolderMatch
    ) {}
    
    get matches(): Match[] {
        return Array.from(this._matches.values());
    }
    
    get count(): number {
        return this._matches.size;
    }
}

export class Match implements ISearchTreeMatch {
    constructor(
        private readonly _range: Range,
        private readonly _text: string,
        private readonly parent: FileMatch
    ) {}
    
    get range(): Range {
        return this._range;
    }
    
    get text(): string {
        return this._text;
    }
    
    get lineNumber(): number {
        return this._range.startLineNumber;
    }
}
```

## Best Practices

<Warning>
**Performance:** Use exclude patterns to skip large directories like `node_modules`. This significantly improves search speed.
</Warning>

<Tip>
**Regex Search:** When using regex, test patterns on a small scope first. Complex regex can be slow on large codebases.
</Tip>

<Note>
**Search Limits:** The default `maxResults` prevents overwhelming results. Increase it carefully for large searches.
</Note>
