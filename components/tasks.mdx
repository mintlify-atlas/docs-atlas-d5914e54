## Overview

The Tasks component (`src/vs/workbench/contrib/tasks/`) provides VS Code's task automation system, allowing you to run build scripts, linters, and other command-line tools from within the editor.

**Location:** `src/vs/workbench/contrib/tasks/`

## Task Service {#task-service}

**Location:** `src/vs/workbench/contrib/tasks/common/taskService.ts`

### ITaskService Interface

```typescript
export interface ITaskService {
    readonly _serviceBrand: undefined;
    
    // Task execution
    run(task: Task | undefined, options?: ITaskRunOptions): Promise<ITaskSummary>;
    
    // Task retrieval
    tasks(filter?: ITaskFilter): Promise<Task[]>;
    getTask(workspaceFolder: IWorkspaceFolder | string, identifier: string): Promise<Task | undefined>;
    
    // Task types
    taskTypes(): string[];
    
    // Configuration
    configureAction(): Action;
    
    // Events
    onDidStateChange: Event<TaskEvent>;
}

export interface Task {
    readonly _id: string;
    readonly type: string;
    readonly configurationProperties: IConfigurationProperties;
    readonly runOptions: IRunOptions;
    
    readonly command: ICommandConfiguration;
    readonly problemMatchers: (string | IProblemMatcher)[];
}
```

### Running Tasks

```typescript
import { ITaskService, Task } from 'vs/workbench/contrib/tasks/common/taskService';

export class TaskRunner {
    constructor(
        @ITaskService private readonly taskService: ITaskService
    ) {}
    
    async runTaskByName(name: string): Promise<void> {
        const tasks = await this.taskService.tasks();
        const task = tasks.find(t => t.configurationProperties.name === name);
        
        if (task) {
            await this.taskService.run(task);
        }
    }
    
    async runBuildTask(): Promise<void> {
        // Run the default build task
        await this.taskService.run(undefined, {
            type: 'build'
        });
    }
    
    async getAllTasks(): Promise<Task[]> {
        return await this.taskService.tasks();
    }
}
```

```json
{
    "problemMatcher": {
        // Ownership
        "owner": "typescript",
        
        // Source
        "source": "ts",

        "fileLocation": [

        // Pattern
        "pattern": {
            "regexp": "^([^\\s].*)[\\(:]+(\\d+)[,:](\\d+)\\)?\\s*:\\s*(error|warning|info)\\s*\\w+\\s*:\\s*(.*)$",
            "file": 1,
            "line": 2,
            "column": 3,
            "severity": 4,
            "message": 5,
            "code": 6,
            "loop": false  // Match multiple times
        },
        
        // Multi-line patterns
        "pattern": [
            {
                "regexp": "^([^\\s].*)",
                "file": 1
            },
            {
                "regexp": "^\\s+(\\d+):(\\d+)\\s+(error|warning)\\s+(.*)$",
                "line": 1,
                "column": 2,
                "severity": 3,
                "message": 4
            }
        ],
        
        // Background task
        "background": {
            "activeOnStart": true,
            "beginsPattern": "^\\s*\\[.*compilation starting",

        // Severity
```

### Auto-detected Tasks

VS Code automatically detects tasks from:
- **npm** - npm scripts from package.json
- **Gulp** - Gulp tasks from gulpfile.js
- **Grunt** - Grunt tasks from Gruntfile.js
- **Jake** - Jake tasks from Jakefile
- **Maven** - Maven goals
- **Gradle** - Gradle tasks

### Custom Task Provider

```typescript
import { Task, ITaskProvider } from 'vs/workbench/contrib/tasks/common/tasks';

class MyTaskProvider implements ITaskProvider {
    static readonly Type = 'mytask';

        // Detect and return tasks
        return [
            {
                type: MyTaskProvider.Type,
                label: 'My Custom Task',
                command: 'echo',
                args: ['Hello']
            }
        ];
    }
    
    async resolveTask(task: Task): Promise<Task | undefined> {
        // Resolve task details
        return task;
    }
}
```

### Task States

```typescript
export enum TaskEventKind {
    Start = 'start',
    ProcessStarted = 'processStarted',
    Active = 'active',
    Inactive = 'inactive',
    Changed = 'changed',
    Terminated = 'terminated',
    ProcessEnded = 'processEnded',
    End = 'end'
}
```

### Monitoring Task Execution

```typescript
import { ITaskService, TaskEvent, TaskEventKind } from 'vs/workbench/contrib/tasks/common/taskService';

export class TaskMonitor {
    constructor(
        @ITaskService private readonly taskService: ITaskService
    ) {
        this.taskService.onDidStateChange(event => {
            this.handleTaskEvent(event);
        });
    }
    
    private handleTaskEvent(event: TaskEvent): void {
        switch (event.kind) {
            case TaskEventKind.Start:
                console.log('Task started:', event.task?.configurationProperties.name);
                break;
            case TaskEventKind.ProcessStarted:
                console.log('Process started, PID:', event.processId);
                break;
            case TaskEventKind.End:
                console.log('Task ended:', event.exitCode);
                break;
            case TaskEventKind.Terminated:
                console.log('Task terminated');
                break;
        }
    }
}
```

## Best Practices

<Warning>
**Problem Matchers:** Test your problem matcher regex carefully. Incorrect patterns can cause performance issues.
</Warning>

<Tip>
**Background Tasks:** Use background tasks for watch modes. They won't block the task system.
</Tip>

<Note>
**Dependencies:** Use task dependencies wisely. Deep dependency chains can be hard to debug.
</Note>

## Extension Integration

### Task Provider Extension

```typescript
// Extension API for task providers
import * as vscode from 'vscode';

class MyTaskProvider implements vscode.TaskProvider {
    async provideTasks(): Promise<vscode.Task[]> {
        const tasks: vscode.Task[] = [];
        
        // Detect tasks
        const task = new vscode.Task(
            { type: 'mytask', task: 'build' },
            vscode.TaskScope.Workspace,
            'Build',
            'mytask',
            new vscode.ShellExecution('npm run build')
        );
        
        task.group = vscode.TaskGroup.Build;
        tasks.push(task);
        
        return tasks;
    }
    
    async resolveTask(task: vscode.Task): Promise<vscode.Task | undefined> {
        return task;
    }
}

// Register provider
vscode.tasks.registerTaskProvider('mytask', new MyTaskProvider());
```
