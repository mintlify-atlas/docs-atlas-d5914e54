### Creating Terminals

```typescript
import { ITerminalService } from 'vs/workbench/contrib/terminal/browser/terminal';
import { ITerminalProfile } from 'vs/platform/terminal/common/terminal';

export class TerminalManager {
    constructor(
        @ITerminalService private readonly terminalService: ITerminalService
    ) {}
    
    async createBasicTerminal(): Promise<void> {
        const terminal = await this.terminalService.createTerminal({
            name: 'My Terminal'
        });
        
        terminal.show();
    }
    
    async createTerminalWithProfile(profile: ITerminalProfile): Promise<void> {
        const terminal = await this.terminalService.createTerminal({
            config: profile,
            cwd: '/workspace'
        });
        
        terminal.show();
    }
    
    async createTerminalWithCommand(): Promise<void> {
        const terminal = await this.terminalService.createTerminal({
            name: 'Build',
            shellArgs: ['-c', 'npm run build']
        });
        
        terminal.show();
    }
    
    getActiveTerminal(): ITerminalInstance | undefined {
        return this.terminalService.activeInstance;
    }
    
    getAllTerminals(): readonly ITerminalInstance[] {
        return this.terminalService.instances;
    }
}
```

### Working with Terminal Instances

```typescript
import { ITerminalInstance } from 'vs/workbench/contrib/terminal/browser/terminal';

export class TerminalController {
    constructor(private readonly terminal: ITerminalInstance) {
        this.setupListeners();
    }
    
    private setupListeners(): void {
        // Listen to title changes
        this.terminal.onTitleChanged(title => {
            console.log('Terminal title:', title);
        });
        
        // Listen to data
        this.terminal.onDataReceived(data => {
            console.log('Terminal output:', data);
        });
        
        // Process ID ready
        this.terminal.onProcessIdReady(pid => {
            console.log('Terminal PID:', pid);
        });
    }
    
    async runCommand(command: string): Promise<void> {
        // Send command and execute
        await this.terminal.sendText(command, true);
    }
    
    async sendTextWithoutExecuting(text: string): Promise<void> {
        // Type text without pressing enter
        await this.terminal.sendText(text, false);
    }
    
    focusTerminal(): void {
        this.terminal.show();
        this.terminal.focus(true);
    }
    
    copySelection(): void {
        if (this.terminal.hasSelection()) {
            // Selection is automatically copied to clipboard
        }
    }
}
```

### Command Detection

```typescript
import { ITerminalInstance } from 'vs/workbench/contrib/terminal/browser/terminal';
import { TerminalCapability } from 'vs/platform/terminal/common/capabilities/capabilities';

function getTerminalCommands(terminal: ITerminalInstance): ITerminalCommand[] {
    const capabilities = terminal.capabilities;
    
    if (capabilities.has(TerminalCapability.CommandDetection)) {
        const commandDetection = capabilities.get(TerminalCapability.CommandDetection);
        return commandDetection.commands;
    }
    
    return [];
}

function onCommandExecuted(terminal: ITerminalInstance): void {
    terminal.capabilities.onDidAddCapability(e => {
        if (e === TerminalCapability.CommandDetection) {
            const commandDetection = terminal.capabilities.get(TerminalCapability.CommandDetection);
            
            commandDetection.onCommandFinished(command => {
                console.log('Command completed:', command.command);
                console.log('Exit code:', command.exitCode);
                console.log('Working directory:', command.cwd);
            });
        }
    });
}
```

### Working with Profiles

```typescript
import { ITerminalService } from 'vs/workbench/contrib/terminal/browser/terminal';
import { ITerminalProfile } from 'vs/platform/terminal/common/terminal';

export class ProfileManager {
    constructor(
        @ITerminalService private readonly terminalService: ITerminalService
    ) {}
    
    async getAvailableProfiles(): Promise<ITerminalProfile[]> {
        return await this.terminalService.getAvailableProfiles();
    }
    
    async createTerminalWithProfile(profileName: string): Promise<void> {
        const profiles = await this.getAvailableProfiles();
        const profile = profiles.find(p => p.profileName === profileName);
        
        if (profile) {
            await this.terminalService.createTerminal({
                config: profile
            });
        }
    }
}
```

### Custom Terminal Commands

```typescript
import { registerAction2, Action2 } from 'vs/platform/actions/common/actions';
import { ITerminalService } from 'vs/workbench/contrib/terminal/browser/terminal';

class RunBuildInTerminal extends Action2 {
    constructor() {
        super({
            id: 'myExtension.runBuild',
            title: { value: 'Run Build', original: 'Run Build' },
            category: { value: 'Terminal', original: 'Terminal' },
            f1: true
        });

        const terminalService = accessor.get(ITerminalService);

            name: 'Build'

        await terminal.sendText('npm run build', true);
    }
}
```

## Best Practices

<Warning>
**Process Management:** Always dispose terminal instances properly. Lingering processes can consume resources.
</Warning>

<Tip>
**Shell Integration:** Enable shell integration for enhanced features like command detection and navigation.
</Tip>

<Note>
**Performance:** Limit scrollback for terminals with heavy output. Large buffers can impact performance.
</Note>
