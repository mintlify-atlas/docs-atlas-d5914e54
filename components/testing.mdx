## Overview

The Testing component (`src/vs/workbench/contrib/testing/`) provides VS Code's unified testing UI, including the Test Explorer view, test execution, and results visualization for any test framework.

### Using Test Service

```typescript
import { ITestService } from 'vs/workbench/contrib/testing/common/testService';

export class TestManager {
    constructor(
        @ITestService private readonly testService: ITestService
    ) {
        // Monitor controllers
        this.testService.onDidRegisterController(controller => {
            console.log('Test controller registered:', controller.label);
        });
    }
    
    async runAllTests(): Promise<void> {
        const collection = this.testService.collection;
        const tests = Array.from(collection.all);
        
        if (tests.length === 0) {
            console.log('No tests found');
            return;
        }
        
        await this.testService.runTests({
            targets: [{
                testIds: tests.map(t => t.item.extId),
                controllerId: tests[0].controllerId,
                profileId: 0
            }],
            exclude: [],
            continuous: false
        });
    }
    
    getTestControllers(): ReadonlyArray<IMainThreadTestController> {
        return this.testService.controllers;
    }
}
```

### Working with Test Collection

```typescript
import { ITestService } from 'vs/workbench/contrib/testing/common/testService';
import { URI } from 'vs/base/common/uri';

export class TestExplorer {
    constructor(
        @ITestService private readonly testService: ITestService
    ) {}
    
    getTestsInFile(uri: URI): IncrementalTestCollectionItem[] {
        const collection = this.testService.collection;
        return Array.from(collection.getNodeByUrl(uri));
    }
    
    async expandTest(testId: string): Promise<void> {
        const collection = this.testService.collection;
        await collection.expand(testId, 1);
    }
    
    getAllTests(): IncrementalTestCollectionItem[] {
        const collection = this.testService.collection;
        return Array.from(collection.all);
    }
    
    getRootTests(): IncrementalTestCollectionItem[] {
        const collection = this.testService.collection;
        return Array.from(collection.rootItems);
    }
}
```

```typescript
import { ITestService } from 'vs/workbench/contrib/testing/common/testService';
import { TestRunProfileBitset } from 'vs/workbench/contrib/testing/common/testTypes';

export class TestRunner {
    constructor(
        @ITestService private readonly testService: ITestService
    ) {}
    
    async runTest(testId: string, profile: TestRunProfileBitset): Promise<void> {
        await this.testService.runTests({
            targets: [{
                testIds: [testId],
                controllerId: this.getControllerId(testId),
                profileId: this.getProfileId(profile)

            exclude: [],
            continuous: false
        });

        await this.runTest(testId, TestRunProfileBitset.Debug);
    }
    
    async runWithCoverage(testId: string): Promise<void> {
        await this.runTest(testId, TestRunProfileBitset.Coverage);
    }
}
```

### Accessing Test Results

```typescript
import { ITestService, ITestResultService } from 'vs/workbench/contrib/testing/common/testService';

export class TestResultViewer {
    constructor(
        @ITestService private readonly testService: ITestService
    ) {}
    
    getLatestResults(): ITestResult[] {
        const results = this.testService.results;
        return results.results;
    }
    
    getTestState(testId: string): TestResultState | undefined {
        const results = this.testService.results;
        const latestResult = results.results[0];
        
        if (latestResult) {
            const item = latestResult.getStateById(testId);
            return item?.state;
        }
        
        return undefined;
    }
    
    getFailedTests(): ITestResultItem[] {
        const results = this.testService.results;
        const latestResult = results.results[0];
        
        if (!latestResult) return [];
        
        return Array.from(latestResult.tests).filter(
            test => test.state === TestResultState.Failed
        );
    }
}
```

### Custom Test Actions

```typescript
import { registerAction2, Action2 } from 'vs/platform/actions/common/actions';
import { ITestService } from 'vs/workbench/contrib/testing/common/testService';

class RunFailedTestsAction extends Action2 {
    constructor() {
        super({
            id: 'myExtension.runFailedTests',
            title: { value: 'Run Failed Tests', original: 'Run Failed Tests' },
            category: { value: 'Test', original: 'Test' },
            f1: true
        });
    }
    
    async run(accessor: ServicesAccessor): Promise<void> {
        const testService = accessor.get(ITestService);
        const results = testService.results;
        const latestResult = results.results[0];
        
        if (!latestResult) return;
        
        const failedTests = Array.from(latestResult.tests)
            .filter(test => test.state === TestResultState.Failed)
            .map(test => test.item.extId);
        
        if (failedTests.length > 0) {
            await testService.runTests({
                targets: [{
                    testIds: failedTests,
                    controllerId: /* ... */,
                    profileId: 0
                }],
                exclude: [],
                continuous: false
            });
        }
    }
}

registerAction2(RunFailedTestsAction);
```

## Extension Integration

### Test Controller Extension

```typescript
// Extension API for test controllers
import * as vscode from 'vscode';

class MyTestController {
    private controller: vscode.TestController;
    
    constructor() {
        // Create controller
        this.controller = vscode.tests.createTestController(
            'myTestController',
            'My Tests'
        );
        
        // Set up discovery
        this.controller.resolveHandler = async (item) => {
            if (!item) {
                // Discover root tests
                await this.discoverTests();
            } else {
                // Discover children
                await this.discoverChildren(item);
            }
        };
        
        // Set up run handler
        const runProfile = this.controller.createRunProfile(
            'Run',
            vscode.TestRunProfileKind.Run,
            async (request, token) => {
                await this.runTests(request, token);
            },
            true // isDefault
        );
    }
    
    private async discoverTests(): Promise<void> {
        // Find test files
        const files = await vscode.workspace.findFiles('**/*.test.js');
        
        for (const file of files) {
            // Create test items
            const testItem = this.controller.createTestItem(
                file.toString(),
                path.basename(file.fsPath),
                file
            );
            
            this.controller.items.add(testItem);
        }
    }
    
    private async runTests(
        request: vscode.TestRunRequest,
        token: vscode.CancellationToken
    ): Promise<void> {
        const run = this.controller.createTestRun(request);
        
        for (const test of request.include ?? this.controller.items) {
            if (token.isCancellationRequested) {
                run.skipped(test);
                continue;
            }
            
            run.started(test);
            
            try {
                // Run test
                const result = await this.executeTest(test);
                
                if (result.passed) {
                    run.passed(test, result.duration);
                } else {
                    const message = new vscode.TestMessage(result.error);
                    run.failed(test, message, result.duration);
                }
            } catch (error) {
                run.errored(test, new vscode.TestMessage(String(error)));
            }
        }
        
        run.end();
    }
}
```
