## Overview

The Workbench (`src/vs/workbench/`) is the application shell that brings together all VS Code components into a cohesive user interface. It manages the layout, views, editors, panels, and overall application lifecycle.

**Location:** `src/vs/workbench/`

### Layout Service

**Location:** `src/vs/workbench/services/layout/browser/layoutService.ts:18`

```typescript
import { IWorkbenchLayoutService } from 'vs/workbench/services/layout/browser/layoutService';

export class MyComponent {
    constructor(
        @IWorkbenchLayoutService private readonly layoutService: IWorkbenchLayoutService
    ) {}
    
    togglePanel(): void {
        const panelVisible = this.layoutService.isVisible(Parts.PANEL_PART);
        this.layoutService.setPartHidden(!panelVisible, Parts.PANEL_PART);
    }
    
    focusEditor(): void {
        this.layoutService.focus(Parts.EDITOR_PART);
    }
    
    getContainer(): HTMLElement {
        return this.layoutService.getContainer(Parts.SIDEBAR_PART);
    }
}
```

```typescript
import { IEditorService } from 'vs/workbench/services/editor/common/editorService';
import { URI } from 'vs/base/common/uri';

export class DocumentOpener {
    constructor(
        @IEditorService private readonly editorService: IEditorService
    ) {}
    
    async openFile(uri: URI): Promise<void> {
        await this.editorService.openEditor({
            resource: uri,
            options: {
                pinned: true,
                selection: { startLineNumber: 10, startColumn: 1 }
            }
        });
    }
    
    async openToSide(uri: URI): Promise<void> {
        await this.editorService.openEditor(
            { resource: uri },
            SIDE_GROUP
        );
    }
    
    getActiveEditor() {
        return this.editorService.activeEditorPane;
    }
}
```

### Lifecycle Phases

<Accordion title="Workbench Lifecycle Phases">

```typescript
export enum LifecyclePhase {
    // Before workbench starts
    Starting = 1,
    
    // Workbench is ready, but not restored
    Ready = 2,
    
    // Workbench state restored, views created
    Restored = 3,
    
    // All extensions activated
    Eventually = 4
}
```

**When to use:**
- **Starting** - Critical initialization that blocks startup
- **Ready** - Early initialization before restoration
- **Restored** - Most contributions (views, commands ready)
- **Eventually** - Non-critical features

</Accordion>

```typescript
import { Action2, registerAction2 } from 'vs/platform/actions/common/actions';
import { ServicesAccessor } from 'vs/platform/instantiation/common/instantiation';
import { IQuickInputService } from 'vs/platform/quickinput/common/quickInput';

class MyAction extends Action2 {
    constructor() {
        super({
            id: 'myExtension.myAction',
            title: { value: 'My Action', original: 'My Action' },
            category: { value: 'My Extension', original: 'My Extension' },
            f1: true, // Show in command palette
            keybinding: {
                weight: 200,
                primary: KeyMod.CtrlCmd | KeyCode.KeyM
            }
        });
    }
    
    async run(accessor: ServicesAccessor): Promise<void> {
        const quickInputService = accessor.get(IQuickInputService);
        const result = await quickInputService.pick(['Option 1', 'Option 2']);
        console.log('Selected:', result);
    }
}

registerAction2(MyAction);
```

## Best Practices

<Warning>
**Contribution Timing:** Register contributions at the appropriate lifecycle phase. Registering too early can slow startup.
</Warning>

<Tip>
**View State:** Persist view state in `IStorageService` to maintain user experience across sessions.
</Tip>

<Note>
**Editor Management:** Always use `IEditorService` to open editors. Direct manipulation can break editor group management.
</Note>
